BT = window.BT || { __namespace: true, __typeName: "BT" };
BT.PCF = BT.PCF || { __namespace: true, __typeName: "BT.PCF" };
BT.PCF.Main =
  BT.PCF.Main ||
  (() => {
    "use strict";
    const Fields = {
      Attributes: {
        Approve: "btr_approve",
        Reject: "btr_reject",
        SingleUpload: "btr_singleupload",
        MultipleUpload: "btr_multipleupload",
      },
    };
    const onload = (executionContext) => {
      const formContext = executionContext.getFormContext();
      if (formContext.ui.getFormType() === 2) {
        formContext.getControl(Fields.Attributes.Approve)?.addEventHandler("mycustomEvent", () => {
          approveButtonHandler(executionContext);
        });
        formContext.getControl(Fields.Attributes.Reject)?.addEventHandler("mycustomEvent", () => {
          rejectButtonHandler(executionContext);
        });
        formContext.getControl(Fields.Attributes.SingleUpload)?.addEventHandler("mycustomEvent", (response) => {
          uploadHandler(executionContext, response, Fields.Attributes.SingleUpload);
        });
        formContext.getControl(Fields.Attributes.MultipleUpload)?.addEventHandler("mycustomEvent", (response) => {
          uploadHandler(executionContext, response, Fields.Attributes.MultipleUpload);
        });
      }
    };
    const uploadHandler = (executionContext, response, field) => {
      const formContext = executionContext.getFormContext();
      const recordId = formContext.data.entity.getId().replace(/[{\}]/g, "");
      Xrm.Utility.showProgressIndicator("Uploading...");
      if (response && response.length > 0) {
        const promises = response.map((file) => {
          var record = {};
          record.documentbody = file.fileContent; // Base64 string
          record.filename = file.fileName; // Text
          record.subject = file.fileName; // Text
          record["objectid_btr_pcfdemo@odata.bind"] = `btr_pcfdemos(${recordId})`;
          return Xrm.WebApi.createRecord("annotation", record);
        });
        Promise.all(promises).then(() => {
          formContext.getAttribute(field)?.setValue("clear");
          formContext.ui.setFormNotification("Upload success", "INFO", "uploadSuccess");
          formContext.getControl("Timeline")?.refresh(); // Refresh timeline to show the new notes/attachments
          Xrm.Utility.closeProgressIndicator();
        });
      }
    };
    const approveButtonHandler = (executionContext) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: "Approved button click.", title: "Alert" });
    };
    const rejectButtonHandler = (executionContext) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: "Rejected button click.", title: "Alert" });
    };
    return {
      OnLoad: onload,
      __namespace: true,
      __typeName: "BT.PCF.Main",
    };
  })();
