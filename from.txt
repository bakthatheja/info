BT = window.BT || { __namespace: true, __typeName: "BT" };
BT.PCF = BT.PCF || { __namespace: true, __typeName: "BT.PCF" };
BT.PCF.Main =
  BT.PCF.Main ||
  (() => {
    "use strict";
    const Fields = {
      Attributes: {
        Approve: "btr_approve",
        Reject: "btr_reject",
        SingleUpload: "btr_singleupload",
        MultipleUpload: "btr_multipleupload",
      },
    };
    const onload = (executionContext) => {
      const formContext = executionContext.getFormContext();
      formContext.getControl(Fields.Attributes.Approve)?.addEventHandler("mycustomEvent", () => {
        approveButtonHandler(executionContext);
      });
      formContext.getControl(Fields.Attributes.Reject)?.addEventHandler("mycustomEvent", () => {
        rejectButtonHandler(executionContext);
      });
      formContext.getControl(Fields.Attributes.SingleUpload)?.addEventHandler("mycustomEvent", (response) => {
        singleUploadButtonHandler(executionContext, response);
      });
      formContext.getControl(Fields.Attributes.MultipleUpload)?.addEventHandler("mycustomEvent", (response) => {
        multipleUploadButtonHandler(executionContext, response);
      });
    };
    const singleUploadButtonHandler = (executionContext, response) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: `single file uploaded: ${response.length}`, title: "Alert" });
    };
    const multipleUploadButtonHandler = (executionContext, response) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: `multiple files uploaded: ${response.length}`, title: "Alert" });
    };
    const approveButtonHandler = (executionContext) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: "Approved button click.", title: "Alert" });
    };
    const rejectButtonHandler = (executionContext) => {
      const formContext = executionContext.getFormContext();
      Xrm.Navigation.openAlertDialog({ confirmButtonLabel: "Ok", text: "Rejected button click.", title: "Alert" });
    };
    return {
      OnLoad: onload,
      __namespace: true,
      __typeName: "BT.PCF.Main",
    };
  })();
